# ==============================================================================
# SilverBullet AppArmor profile for Home Assistant
#
# Reference: https://developers.home-assistant.io/docs/add-ons/security#apparmor
#
# Note: The Home Assistant Supervisor automatically replaces the profile name
# with the actual add-on name. A space_path outside /share/silverbullet/ is
# blocked by the profile.
# ==============================================================================
#include <tunables/global>

profile hassio-addon-silverbullet flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  # ─── Capabilities ──────────────────────────────────────────────────────────
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,
  capability sys_chroot,
  capability net_bind_service,

  # ─── Signals ───────────────────────────────────────────────────────────────
  signal (send, receive) peer=unconfined,
  signal (send, receive) peer=hassio-addon-silverbullet,
  # docker stop / the Supervisor send SIGTERM; the sender (runc) has its own label depending on the host
  signal (receive) set=(term, int, hup, kill),

  # ─── ptrace for Chromium child processes ───────────────────────────────────────
  ptrace (read, trace) peer=hassio-addon-silverbullet,

  # ─── Network access ──────────────────────────────────────────────────────
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  network unix dgram,
  network netlink raw,

  # ─── s6-overlay and init paths ────────────────────────────────────────────
  # s6-overlay v3 lives under /package/admin/<tool>-<version>/command/ and
  # starts its scripts from /run/s6*, /etc/s6-overlay and /etc/cont-*.d.
  /init                        rix,
  /package/**                  rix,
  /command/**                  rix,
  /run/s6/**                   rix,
  /run/s6-rc*/**               rix,
  /run/service/**              rix,
  /etc/s6-overlay/**           rwix,
  /etc/cont-init.d/**          rwix,
  /etc/cont-finish.d/**        rwix,
  /etc/services.d/**           rwix,
  /etc/fix-attrs.d/**          r,

  # ─── System binaries (read + execute) ─────────────────────────────────────
  /bin/**                      rix,
  /sbin/**                     rix,
  /usr/bin/**                  rix,
  /usr/sbin/**                 rix,
  /usr/local/bin/**            rix,
  /usr/local/sbin/**           rix,

  # ─── Libraries (read + mmap) ───────────────────────────────────────────
  /lib/**                      rm,
  /lib64/**                    rm,
  /usr/lib/**                  rm,
  /usr/local/lib/**            rm,

  # ─── Chromium (headless Runtime API for SilverBullet AI plugs) ────────────
  /usr/lib/chromium/**         mrix,

  # ─── Node.js modules ───────────────────────────────────────────────────────
  # silverbullet-ai-mcp is a Node script (shebang) and must be executable
  /usr/lib/node_modules/**     rmix,
  /usr/local/lib/node_modules/** rmix,

  # ─── Add-on binaries and scripts ──────────────────────────────────────────
  /run.sh                      rix,
  /usr/local/bin/silverbullet  rix,
  /usr/local/bin/silverbullet-ai-mcp rix,
  # bashio is the interpreter of run.sh (#!/usr/bin/with-contenv bashio)
  /usr/lib/bashio/**           rix,

  # ─── HOME directory for Chromium, Node/npm, SilverBullet ────────────────
  /root/                       r,
  owner /root/**              rwk,

  # ─── Data and storage paths ─────────────────────────────────────────────
  /data/**                     rwk,
  /share/ r,
  /share/silverbullet{,/**}    rwk,
  # Chromium font cache (fontconfig)
  /var/cache/fontconfig/       r,
  /var/cache/fontconfig/**     rwkl,
  /tmp/                        r,
  /tmp/**                      rwkm,
  /run/                        rw,
  /run/**                      rwk,

  # ─── System and device paths ─────────────────────────────────────────────
  /etc/**                      r,
  /usr/share/**                r,
  /dev/null                    rw,
  /dev/zero                    r,
  /dev/full                    rw,
  /dev/random                  r,
  /dev/urandom                 r,
  /dev/tty                     rw,
  /dev/console                 rw,
  /dev/pts/**                  rw,
  /dev/shm/**                  rwkm,

  # ─── Proc access ─────────────────────────────────────────────────────────
  @{PROC}/                     r,
  @{PROC}/self/**              r,
  @{PROC}/@{pid}/**            r,
  @{PROC}/@{pid}/oom_score_adj rw,
  @{PROC}/*/stat               r,
  @{PROC}/*/status             r,
  @{PROC}/sys/kernel/**        r,
  @{PROC}/sys/fs/**            r,
  @{PROC}/sys/vm/**            r,
  @{PROC}/meminfo              r,
  @{PROC}/uptime               r,
  @{PROC}/version_signature    r,
  @{PROC}/cpuinfo              r,
  @{PROC}/stat                 r,
  @{PROC}/loadavg              r,
  @{PROC}/filesystems          r,
  @{PROC}/mounts               r,

  # ─── System information ────────────────────────────────────────────────
  /sys/**                      r,

  # ─── Explicit denies ────────────────────────────────────────────────────
  deny /etc/shadow             rw,
  deny /etc/gshadow            rw,
  deny /root/.ssh/**           rw,
  deny /proc/sysrq-trigger     rw,
  deny /sys/kernel/**          rw,
}
